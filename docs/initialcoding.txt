"""
wifi_mapper.py

Single-file merge of:
  - logger.py      (serial -> Google Sheets logger)
  - data.py        (fetch/clean/cluster)
  - heat_layer.py  (heatmap layer)
  - point_layers.py(per-SSID point layers + select-all control)
  - controls.py    (Point Map / Heatmap view toggle)
  - main.py        (orchestration)

Usage:
    python wifi_mapper.py log      # run the serial -> Google Sheets logger.
                                    # Also starts a local live map at
                                    # http://localhost:8765 that updates
                                    # instantly per-row, in parallel with
                                    # the batched Sheet writes — so the
                                    # sheet ("Excel") and the map update
                                    # together from the same read loop.
    python wifi_mapper.py map      # fetch data from the sheet and build a static wifi_map.html
    python wifi_mapper.py live     # generate a self-refreshing live_map.html that polls the
                                    # Sheet's CSV export instead (useful if 'log' is running on
                                    # a different machine than the one viewing the map)
    python wifi_mapper.py reset    # wipe all logged rows from the sheet + delete stale local map files
                                    # (asks "Type YES to confirm"; add --yes to skip the prompt)
    python wifi_mapper.py          # defaults to "map"

Note: "log", "map", and "live" are independent workflows. "log" is the only
one that touches the serial port. "map" builds a one-time static snapshot
from whatever is in the sheet right now. "live" never touches the serial
port at all — it polls the same Google Sheet in the browser every few
seconds and moves a marker to the latest GPS fix, so it's safe to run
alongside "log" at the same time to watch your position update live.
"""

def _has_valid_gps_fix(lat_str, lon_str) -> bool:
    """
    True only if both values parse as floats and aren't the (0, 0) / blank
    placeholder a GPS module reports before it has acquired a fix.
    """
    try:
        lat = float(lat_str)
        lon = float(lon_str)
    except (ValueError, TypeError):
        return False
    if lat == 0.0 and lon == 0.0:
        return False
    return True

HEADERS = [
    "Timestamp", "Latitude", "Longitude", "Altitude", "Satellites",
    "SSID", "RSSI_dBm", "Channel", "Encryption", "SignalQuality",
]


def _authorize_sheet():
    """Authorize with the service account and return the sheet1 worksheet."""
    import gspread
    from oauth2client.service_account import ServiceAccountCredentials

    scope = [
        "https://spreadsheets.google.com/feeds",
        "https://www.googleapis.com/auth/drive",
    ]
    creds = ServiceAccountCredentials.from_json_keyfile_name("credentials.json", scope)
    client = gspread.authorize(creds)
    return client.open_by_key(SHEET_ID).sheet1


def _setup_headers(sheet):
    """Write headers if missing, then bold them and freeze the row."""
    if not sheet.get_all_values():
        sheet.append_row(HEADERS)
    sheet.format("A1:J1", {"textFormat": {"bold": True}})
    sheet.freeze(rows=1)


def reset_all_data(delete_local_files: bool = True, skip_confirm: bool = False) -> None:
    """
    Wipe every logged row from the Google Sheet (keeping only the header),
    and delete any stale local map files so a fresh `log` run starts from
    zero surrounding networks instead of showing everything ever logged.

    This is destructive and irreversible, so it asks for confirmation
    unless skip_confirm=True (e.g. `python wifi_mapper.py reset --yes`).
    """
    if not skip_confirm:
        answer = input(
            "This will permanently delete ALL logged rows from the Google "
            "Sheet and any local wifi_map.html / live_map.html files.\n"
            "Type YES to confirm: "
        )
        if answer.strip() != "YES":
            print("Reset cancelled. Nothing was deleted.")
            return

    print("Connecting to Google Sheet...")
    sheet = _authorize_sheet()

    print("Clearing all rows...")
    sheet.clear()
    _setup_headers(sheet)
    print("Sheet cleared — only the header row remains.")

    if delete_local_files:
        for fname in ("wifi_map.html", "live_map.html"):
            if os.path.exists(fname):
                os.remove(fname)
                print(f"Deleted stale local file: {fname}")

    print("Reset complete. Run 'log' again to start collecting fresh data —")
    print("only scans taken after the GPS has a real fix will be recorded.")

# In-memory snapshot of "what's around me right now" — updated the instant a
# valid-fix row is read, independent of the batched Sheet flush. This is what
# makes the map update in real time instead of waiting on the Sheets round-trip.
_live_state_lock = threading.Lock()
_live_state = {
    "timestamp": None,
    "latitude": None,
    "longitude": None,
    "altitude": None,
    "satellites": None,
    "networks": [],  # networks seen at the CURRENT fix (cleared when lat/lon changes)
}


def _update_live_state(ts, lat, lon, alt, sats, ssid, rssi, channel, encryption, quality):
    """Update the in-memory snapshot the local live map reads from.
    Networks list resets whenever the GPS fix moves to a new coordinate,
    so it always reflects only the networks around the current location."""
    with _live_state_lock:
        same_fix = (_live_state["latitude"] == lat and _live_state["longitude"] == lon)
        if not same_fix:
            _live_state["networks"] = []

        _live_state["timestamp"] = ts
        _live_state["latitude"] = lat
        _live_state["longitude"] = lon
        _live_state["altitude"] = alt
        _live_state["satellites"] = sats

        _live_state["networks"] = [n for n in _live_state["networks"] if n["ssid"] != ssid]
        _live_state["networks"].append({
            "ssid": ssid, "rssi": rssi, "channel": channel,
            "encryption": encryption, "quality": quality,
        })


def _print_live_status_loop(interval_seconds: int = 5):
    """
    Runs forever in a background thread: every `interval_seconds`, prints a
    consolidated snapshot of the current fix + networks seen there to the
    terminal — independent of the per-row prints in the main read loop, so
    the terminal keeps showing live output even during quiet stretches.
    """
    while True:
        time.sleep(interval_seconds)
        with _live_state_lock:
            ts = _live_state["timestamp"]
            lat = _live_state["latitude"]
            lon = _live_state["longitude"]
            alt = _live_state["altitude"]
            sats = _live_state["satellites"]
            networks = list(_live_state["networks"])

        print("\n" + "=" * 70)
        if lat is None:
            print("LIVE STATUS: waiting for first GPS fix...")
        else:
            print(f"LIVE STATUS @ {ts}")
            print(f"Location: {lat:.6f}, {lon:.6f}  (alt {alt} m, {sats} satellites)")
            print(f"Networks visible at this location ({len(networks)}):")
            for n in networks:
                print(f"   {n['ssid']:<28} {n['rssi']:>5} dBm  ch{n['channel']:<3} {n['encryption']}")
        print("=" * 70)

class _LiveRequestHandler(http.server.BaseHTTPRequestHandler):
    def do_GET(self):
        if self.path.startswith("/latest"):
            with _live_state_lock:
                payload = json.dumps(_live_state).encode("utf-8")
            self.send_response(200)
            self.send_header("Content-Type", "application/json")
            self.send_header("Access-Control-Allow-Origin", "*")
            self.end_headers()
            self.wfile.write(payload)
        else:
            body = LOCAL_LIVE_HTML.encode("utf-8")
            self.send_response(200)
            self.send_header("Content-Type", "text/html")
            self.end_headers()
            self.wfile.write(body)

    def log_message(self, format, *args):
        pass  # silence per-request console spam


def _start_local_live_server(port: int = LOCAL_SERVER_PORT):
    server = http.server.ThreadingHTTPServer(("0.0.0.0", port), _LiveRequestHandler)
    thread = threading.Thread(target=server.serve_forever, daemon=True)
    thread.start()
    return server


def run_logger(port: str = "COM6", baud: int = 115200):
    """Continuously read WiFi scan lines from a serial port, and:
      1. Batch-append valid rows to the Google Sheet ("Excel"), retrying on rate limits.
      2. Simultaneously update an in-process live map (localhost) instantly per row,
         so the sheet and the map both reflect the current location's networks
         at the same time, from the same read loop.

    Lines are only accepted once the GPS module has an actual fix — any scan
    line arriving before that (typically carrying 0,0 or blank coordinates)
    is dropped rather than written to the sheet or the live map.
    """
    from gspread.exceptions import APIError
    import serial

    sheet = _authorize_sheet()

    IST = timezone(timedelta(hours=5, minutes=30))
    FLUSH_INTERVAL = 5
    MAX_RETRIES = 5

    _setup_headers(sheet)

    def flush_batch(batch):
        if not batch:
            return
        delay = 2
        for attempt in range(MAX_RETRIES):
            try:
                sheet.append_rows(batch, value_input_option="RAW")
                print(f"Flushed {len(batch)} rows to Sheets.")
                return
            except APIError as e:
                if "429" in str(e) or "Quota exceeded" in str(e):
                    print(f"Rate limited, retrying in {delay}s... (attempt {attempt + 1}/{MAX_RETRIES})")
                    time.sleep(delay)
                    delay *= 2
                else:
                    print(f"Sheets API error: {e}")
                    return
        print(f"Failed to flush {len(batch)} rows after {MAX_RETRIES} retries. Data kept for next flush.")

    setup_headers()

    _start_local_live_server()
    print(f"Live map running at http://localhost:{LOCAL_SERVER_PORT} — "
          f"open it in a browser now. It shows ONLY the networks from the "
          f"current scan, redrawn fresh every 5s — not every network ever "
          f"logged to the sheet.")

    threading.Thread(target=_print_live_status_loop, args=(5,), daemon=True).start()
    print("Terminal will print a live status summary every 5 seconds.")
    print("This is a continuous tracking loop — it runs until you stop it with Ctrl+C.")

    ser = serial.Serial(port, baud, timeout=1)
    print("Logging live to Google Sheets (batched, IST timestamps)...")

    batch = []
    last_flush = time.time()

    try:
        while True:
            line = ser.readline().decode(errors="ignore").strip()
            if line and "," in line and not line.startswith("Timestamp"):
                row = line.split(",")
                if len(row) == 10:
                    lat_str, lon_str = row[1], row[2]
                    if not _has_valid_gps_fix(lat_str, lon_str):
                        # No GPS fix yet — don't log this WiFi network at all,
                        # and don't feed it to the live map either.
                        continue
                    ist_time = datetime.now(IST).strftime("%Y-%m-%d %H:%M:%S")
                    row[0] = ist_time
                    batch.append(row)
                    print(row)

                    # Update the live map instantly (independent of the
                    # batched Sheets flush below), so map + sheet both
                    # reflect the current fix's networks simultaneously.
                    _update_live_state(
                        ts=row[0], lat=float(row[1]), lon=float(row[2]),
                        alt=row[3], sats=row[4], ssid=row[5], rssi=row[6],
                        channel=row[7], encryption=row[8], quality=row[9],
                    )

            if batch and (time.time() - last_flush) >= FLUSH_INTERVAL:
                flush_batch(batch)
                batch = []
                last_flush = time.time()

    except KeyboardInterrupt:
        print("\nStopped. Flushing remaining rows...")
        flush_batch(batch)
    finally:
        ser.close()

def fetch_data(url: str = SHEET_URL) -> pd.DataFrame:
    return pd.read_csv(url)


def clean_data(df: pd.DataFrame) -> pd.DataFrame:
    df = df.dropna(subset=["Latitude", "Longitude", "RSSI_dBm"])
    # Drop rows logged before the GPS had a fix (placeholder 0,0 coordinates).
    df = df[~((df["Latitude"] == 0) & (df["Longitude"] == 0))]
    return df


def cluster_by_ssid(df: pd.DataFrame) -> pd.DataFrame:
    """Group nearby measurements per SSID into single points."""
    grouped_data = []

    for ssid, ssid_df in df.groupby("SSID"):
        coords = np.radians(ssid_df[["Latitude", "Longitude"]].values)

        db = DBSCAN(
            eps=DISTANCE_THRESHOLD / EARTH_RADIUS,
            min_samples=1,
            metric="haversine",
        )
        labels = db.fit_predict(coords)

        ssid_df = ssid_df.copy()
        ssid_df["cluster"] = labels

        for _, cluster in ssid_df.groupby("cluster"):
            grouped_data.append({
                "Latitude": cluster["Latitude"].mean(),
                "Longitude": cluster["Longitude"].mean(),
                "SSID": ssid,
                "RSSI_dBm": cluster["RSSI_dBm"].median(),
                "Channel": cluster["Channel"].mode().iloc[0],
                "Encryption": cluster["Encryption"].mode().iloc[0],
                "SignalQuality": cluster["SignalQuality"].mode().iloc[0],
                "Altitude": cluster["Altitude"].mean(),
                "Measurements": len(cluster),
                "Timestamp": cluster["Timestamp"].iloc[0],
            })

    return pd.DataFrame(grouped_data)

def normalize_rssi(rssi: float) -> float:
    """Map RSSI (dBm) to a 0-1 weight, clipped to [RSSI_MIN, RSSI_MAX]."""
    clipped = np.clip(rssi, RSSI_MIN, RSSI_MAX)
    return (clipped - RSSI_MIN) / (RSSI_MAX - RSSI_MIN)


def build_heat_layer(df: pd.DataFrame, name: str = "Signal Heatmap") -> folium.FeatureGroup:
    """
    Build a HeatMap layer wrapped in a FeatureGroup so it behaves like
    the other toggleable layers (has .get_name(), can be add/removed).

    Expects raw (unclustered) rows — don't feed this the DBSCAN-clustered
    output from cluster_by_ssid, since averaging nearby points throws away
    the density signal a heatmap is meant to show.
    """
    heat_points = [
        [row["Latitude"], row["Longitude"], normalize_rssi(row["RSSI_dBm"])]
        for _, row in df.iterrows()
    ]

    heat_group = folium.FeatureGroup(name=name, show=False)

    HeatMap(
        heat_points,
        radius=18,
        blur=22,
        max_zoom=19,
        min_opacity=0.3,
    ).add_to(heat_group)

    return heat_group

def rssi_color(rssi: float) -> str:
    if rssi >= -50:
        return "green"
    elif rssi >= -60:
        return "lightgreen"
    elif rssi >= -70:
        return "orange"
    elif rssi >= -80:
        return "red"
    else:
        return "darkred"


def build_ssid_layers(df: pd.DataFrame, m: folium.Map) -> dict:
    """Create one FeatureGroup per SSID, populated with markers, added to m."""
    ssid_layers = {}

    for ssid in sorted(df["SSID"].unique()):
        ssid_layers[ssid] = folium.FeatureGroup(name=ssid, show=True)
        ssid_layers[ssid].add_to(m)

    for _, row in df.iterrows():
        lat, lon, rssi, ssid = row["Latitude"], row["Longitude"], row["RSSI_dBm"], row["SSID"]

        popup_text = f"""
        <b>SSID:</b> {row["SSID"]}<br>
        <b>RSSI:</b> {rssi} dBm<br>
        <b>Channel:</b> {row["Channel"]}<br>
        <b>Encryption:</b> {row["Encryption"]}<br>
        <b>Signal Quality:</b> {row["SignalQuality"]}<br>
        <b>Latitude:</b> {lat}<br>
        <b>Longitude:</b> {lon}<br>
        <b>Altitude:</b> {row["Altitude"]} m<br>
        <b>Timestamp:</b> {row["Timestamp"]}
        """

        folium.CircleMarker(
            location=[lat, lon],
            radius=6,
            color="black",
            weight=1,
            fill=True,
            fill_color=rssi_color(rssi),
            fill_opacity=0.8,
            popup=folium.Popup(popup_text, max_width=300),
        ).add_to(ssid_layers[ssid])

    return ssid_layers

            checkbox.indeterminate = !initiallyAllChecked && !initiallyNoneChecked;

            checkbox.addEventListener("change", function() {{
                layerInputs.forEach(function(input, idx) {{
                    const layer = window[layerVars[idx]];
                    if (!layer || !mapObj) return;

                    input.checked = checkbox.checked;

                    if (checkbox.checked) {{
                        if (!mapObj.hasLayer(layer)) mapObj.addLayer(layer);
                    }} else {{
                        if (mapObj.hasLayer(layer)) mapObj.removeLayer(layer);
                    }}
                }});
            }});

            layerInputs.forEach(function(input) {{
                input.addEventListener("change", function() {{
                    const allSelected = Array.from(layerInputs).every((i) => i.checked);
                    const noneSelected = Array.from(layerInputs).every((i) => !i.checked);
                    checkbox.checked = allSelected;
                    checkbox.indeterminate = !allSelected && !noneSelected;
                }});
            }});

        }}, 500);
    }});
    </script>
    """

    m.get_root().html.add_child(Element(select_all_html))


# ============================================================
# controls.py — Point Map / Heatmap view toggle
# ============================================================

def inject_view_toggle(m: folium.Map, ssid_layers: dict, heat_layer: folium.FeatureGroup) -> None:
    """
    Add a Point Map / Heatmap toggle control (top-left, below zoom).
    Switching to Heatmap hides all SSID layers + the layer control panel.
    Switching back restores each SSID layer per its checkbox's last state.
    """
    map_var_name = m.get_name()
    heat_var_name = heat_layer.get_name()
    sorted_ssids = sorted(ssid_layers.keys())
    ssid_layer_vars_json = json.dumps([ssid_layers[s].get_name() for s in sorted_ssids])

    toggle_html = f"""
    <script>
    document.addEventListener("DOMContentLoaded", function() {{
        setTimeout(function() {{
            const mapObj = window["{map_var_name}"];
            const heatLayer = window["{heat_var_name}"];
            const ssidLayerVars = {ssid_layer_vars_json};

            if (mapObj.hasLayer(heatLayer)) mapObj.removeLayer(heatLayer);

            const layerControlContainer = document.querySelector('.leaflet-control-layers');

            const ViewToggle = L.Control.extend({{
                options: {{ position: 'topleft' }},
                onAdd: function() {{
                    const container = L.DomUtil.create('div', 'leaflet-bar view-toggle-control');
                    container.style.background = 'white';
                    container.style.padding = '4px';

                    const pointBtn = L.DomUtil.create('a', 'view-toggle-btn active', container);
                    pointBtn.innerHTML = 'Point Map';
                    pointBtn.href = '#';

                    const heatBtn = L.DomUtil.create('a', 'view-toggle-btn', container);
                    heatBtn.innerHTML = 'Heatmap';
                    heatBtn.href = '#';

                    L.DomEvent.disableClickPropagation(container);

                    L.DomEvent.on(pointBtn, 'click', function(e) {{
                        L.DomEvent.preventDefault(e);
                        if (mapObj.hasLayer(heatLayer)) mapObj.removeLayer(heatLayer);

                        const checkboxes = document.querySelectorAll(
                            '.leaflet-control-layers-overlays input[type="checkbox"]'
                        );
                        checkboxes.forEach(function(cb, idx) {{
                            const layer = window[ssidLayerVars[idx]];
                            if (!layer) return;
                            if (cb.checked && !mapObj.hasLayer(layer)) mapObj.addLayer(layer);
                        }});

                        if (layerControlContainer) layerControlContainer.style.display = '';
                        pointBtn.classList.add('active');
                        heatBtn.classList.remove('active');
                    }});

                    L.DomEvent.on(heatBtn, 'click', function(e) {{
                        L.DomEvent.preventDefault(e);
                        ssidLayerVars.forEach(function(varName) {{
                            const layer = window[varName];
                            if (layer && mapObj.hasLayer(layer)) mapObj.removeLayer(layer);
                        }});
                        if (!mapObj.hasLayer(heatLayer)) mapObj.addLayer(heatLayer);

                        if (layerControlContainer) layerControlContainer.style.display = 'none';
                        heatBtn.classList.add('active');
                        pointBtn.classList.remove('active');
                    }});

                    return container;
                }}
            }});

            mapObj.addControl(new ViewToggle());

        }}, 500);
    }});
    </script>
<style>
.view-toggle-control {{
    width: auto !important;
}}
.view-toggle-btn {{
    display: block;
    width: auto !important;
    height: auto !important;
    line-height: normal !important;
    padding: 6px 12px;
    text-decoration: none;
    color: #333;
    font-size: 13px;
    white-space: nowrap;
}}
.view-toggle-btn.active {{
    background: #e6e6e6;
    font-weight: bold;
}}
</style>
    """

    m.get_root().html.add_child(Element(toggle_html))


# ============================================================
# live_view.py — live GPS position tracker (new)
# ============================================================
#
# This does NOT read the serial port directly (the `log` process already
# owns it). Instead it generates a self-refreshing HTML page that polls
# the same Google Sheet the logger is writing to, grabs the most recent
# row's Latitude/Longitude, and moves a marker there — so you can run
# `log` and `live` at the same time and watch your position update on
# the map as new rows land in the sheet.

LIVE_MAP_TEMPLATE = """<!DOCTYPE html>
<html>
<head>
<meta charset="utf-8">
<title>Live GPS Tracker</title>
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<link rel="stylesheet" href="https://unpkg.com/leaflet@1.9.4/dist/leaflet.css" />
<script src="https://unpkg.com/leaflet@1.9.4/dist/leaflet.js"></script>
<script src="https://cdnjs.cloudflare.com/ajax/libs/PapaParse/5.4.1/papaparse.min.js"></script>
<style>
  html, body {{ height: 100%; margin: 0; }}
  #map {{ height: 100%; width: 100%; }}
  #info {{
    position: absolute; top: 10px; right: 10px; z-index: 1000;
    background: white; padding: 10px 14px; border-radius: 6px;
    box-shadow: 0 1px 4px rgba(0,0,0,0.3);
    font-family: sans-serif; font-size: 13px; line-height: 1.5;
    min-width: 200px;
  }}
  #info b {{ display: inline-block; width: 90px; }}
  #status-dot {{
    height: 10px; width: 10px; border-radius: 50%; display: inline-block;
    margin-right: 6px; background: #999;
  }}
  #status-dot.live {{ background: #2ecc71; }}
  #status-dot.stale {{ background: #e74c3c; }}
</style>
</head>
<body>
<div id="map"></div>
<div id="info">
  <div><span id="status-dot"></span><b id="status-text">Connecting...</b></div>
  <div><b>Timestamp:</b> <span id="f-time">-</span></div>
  <div><b>Lat/Lon:</b> <span id="f-latlon">-</span></div>
  <div><b>Altitude:</b> <span id="f-alt">-</span></div>
  <div><b>Satellites:</b> <span id="f-sats">-</span></div>
  <div><b>Last SSID:</b> <span id="f-ssid">-</span></div>
</div>

<script>
const CSV_URL = "{sheet_url}";
const REFRESH_MS = {refresh_ms};
const STALE_AFTER_MS = {stale_after_ms};

const map = L.map('map').setView([0, 0], 3);
L.tileLayer('https://{{s}}.tile.openstreetmap.org/{{z}}/{{x}}/{{y}}.png', {{
    attribution: '&copy; OpenStreetMap contributors',
    maxZoom: 20,
}}).addTo(map);

let marker = null;
let trail = L.polyline([], {{ color: '#3388ff', weight: 3 }}).addTo(map);
let hasCentered = false;
let lastTimestamp = null;

const pulseIcon = L.divIcon({{
    className: '',
    html: '<div style="width:16px;height:16px;border-radius:50%;background:#3388ff;' +
          'border:3px solid white;box-shadow:0 0 6px rgba(0,0,0,0.5);"></div>',
    iconSize: [16, 16],
    iconAnchor: [8, 8],
}});

function setStatus(state, text) {{
    const dot = document.getElementById('status-dot');
    const label = document.getElementById('status-text');
    dot.className = state;
    label.textContent = text;
}}

async function refresh() {{
    try {{
        const res = await fetch(CSV_URL + "&_=" + Date.now(), {{ cache: "no-store" }});
        if (!res.ok) throw new Error("HTTP " + res.status);
        const text = await res.text();

        const parsed = Papa.parse(text.trim(), {{ header: true, skipEmptyLines: true }});
        const rows = parsed.data;
        if (!rows.length) throw new Error("No rows yet");

        const last = rows[rows.length - 1];
        const lat = parseFloat(last.Latitude);
        const lon = parseFloat(last.Longitude);
        if (isNaN(lat) || isNaN(lon) || (lat === 0 && lon === 0)) {{
            throw new Error("No GPS fix yet in latest row");
        }}

        if (!marker) {{
            marker = L.marker([lat, lon], {{ icon: pulseIcon }}).addTo(map);
        }} else {{
            marker.setLatLng([lat, lon]);
        }}
        trail.addLatLng([lat, lon]);

        if (!hasCentered) {{
            map.setView([lat, lon], 18);
            hasCentered = true;
        }} else {{
            map.panTo([lat, lon]);
        }}

        document.getElementById('f-time').textContent = last.Timestamp || "-";
        document.getElementById('f-latlon').textContent = lat.toFixed(6) + ", " + lon.toFixed(6);
        document.getElementById('f-alt').textContent = (last.Altitude || "-") + " m";
        document.getElementById('f-sats').textContent = last.Satellites || "-";
        document.getElementById('f-ssid').textContent = last.SSID || "-";

        const isNewFix = last.Timestamp !== lastTimestamp;
        lastTimestamp = last.Timestamp;
        setStatus(isNewFix ? 'live' : 'stale',
                   isNewFix ? 'Live' : 'No new fix yet');
    }} catch (err) {{
        setStatus('stale', 'Error: ' + err.message);
        console.error(err);
    }}
}}

refresh();
setInterval(refresh, REFRESH_MS);
</script>
</body>
</html>
"""


def build_live_map(
    out_path: str = "live_map.html",
    refresh_seconds: int = 5,
    stale_after_seconds: int = 30,
) -> None:
    """
    Generate a self-refreshing HTML page that shows the most recent GPS
    fix as a moving marker + trail, by polling the same Google Sheet the
    `log` mode writes to. Safe to run at the same time as `log` mode,
    since it never touches the serial port.
    """
    html = LIVE_MAP_TEMPLATE.format(
        sheet_url=SHEET_URL,
        refresh_ms=refresh_seconds * 1000,
        stale_after_ms=stale_after_seconds * 1000,
    )
    with open(out_path, "w", encoding="utf-8") as f:
        f.write(html)

    print(f"Live tracker created: {out_path}")
    print(f"Open it in a browser while 'log' mode is running — "
          f"it refreshes every {refresh_seconds}s.")


# ============================================================
# main.py — orchestration
# ============================================================

def build_map():
    raw_df = fetch_data()
    df = clean_data(raw_df)

    clustered_df = cluster_by_ssid(df)   # for point markers
    heatmap_df = df                       # raw, unclustered — per earlier decision

    center_lat = clustered_df["Latitude"].mean()
    center_lon = clustered_df["Longitude"].mean()

    m = folium.Map(
        location=[center_lat, center_lon],
        zoom_start=18,
        tiles="OpenStreetMap",
    )

    folium.TileLayer(
        tiles="https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}",
        attr="Esri, Maxar, Earthstar Geographics",
        name="Satellite",
        overlay=False,
        control=True,
    ).add_to(m)

    # Point map layers (one FeatureGroup per SSID)
    ssid_layers = build_ssid_layers(clustered_df, m)

    # Heatmap layer (hidden by default)
    heat_layer = build_heat_layer(heatmap_df)
    heat_layer.add_to(m)

    # Layer control (SSID checkboxes)
    folium.LayerControl(collapsed=False).add_to(m)

    # Select-all checkbox wiring
    inject_select_all_script(m, ssid_layers)

    # Point Map / Heatmap toggle
    inject_view_toggle(m, ssid_layers, heat_layer)

    m.save("wifi_map.html")
    print("Map created successfully!")
    print("Open wifi_map.html in your browser.")


def main():
    mode = sys.argv[1] if len(sys.argv) > 1 else "map"
    skip_confirm = "--yes" in sys.argv or "-y" in sys.argv

    if mode == "log":
        run_logger()
    elif mode == "map":
        build_map()
    elif mode == "live":
        build_live_map()
    elif mode == "reset":
        reset_all_data(skip_confirm=skip_confirm)
    else:
        print(f"Unknown mode '{mode}'. Use 'log', 'map', 'live', or 'reset'.")
        sys.exit(1)


if __name__ == "__main__":
    main()
